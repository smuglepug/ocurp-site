  function num(x, d){ if(x === null || x === undefined || !isFinite(x)) return '-';
    return String(parseFloat(x.toFixed(d === undefined ? 2 : d))); }
  function raw(s){ var x = parseFloat(String(s).replace(/[^0-9.\-eE+]/g,'')); return isFinite(x) ? x : NaN; }
  function esc(s){ return String(s).replace(/&/g,'&').replace(/= list[i][1] && pct <= list[i][2]) { out = list[i][0]; break; }
    }
    return out;
  }

  function compute(inp){
    var male = (inp.sex === 'm');
    var pct = male ? navyMale(inp.waist, inp.neck, inp.height)
                   : navyFemale(inp.waist, inp.hip, inp.neck, inp.height);
    var fat = inp.weight * pct / 100;
    return {
      sex: inp.sex, weight: inp.weight, height: inp.height,
      neck: inp.neck, waist: inp.waist, hip: inp.hip,
      body_fat_pct: pct, fat_mass: fat, lean_mass: inp.weight - fat,
      band: band(inp.sex, pct)
    };
  }

  function text(c){
    var sum = (c.sex === 'm')
      ? 'waist - neck = ' + num(c.waist - c.neck, 1) + ' cm'
      : 'waist + hip - neck = ' + num(c.waist + c.hip - c.neck, 1) + ' cm';
    return [
      'BODY FAT - US NAVY',
      '------------------',
      'Sex            ' + (c.sex === 'm' ? 'Male' : 'Female'),
      'Weight         ' + num(c.weight,1) + ' kg',
      'Height         ' + num(c.height,1) + ' cm',
      'Neck           ' + num(c.neck,1) + ' cm',
      'Waist          ' + num(c.waist,1) + ' cm',
      'Hip            ' + num(c.hip,1) + ' cm',
      'Girth sum      ' + sum,
      '',
      'Body fat       ' + num(c.body_fat_pct,2) + ' %',
      'Fat mass       ' + num(c.fat_mass,2) + ' kg',
      'Lean mass      ' + num(c.lean_mass,2) + ' kg',
      'Fitness band   ' + c.band
    ].join('\n');
  }

  function render(c){
    $('f-pct').textContent = num(c.body_fat_pct, 1) + ' %';
    $('f-band').textContent = c.band;
    $('f-fat').textContent  = num(c.fat_mass, 2) + ' kg';
    $('f-lean').textContent = num(c.lean_mass, 2) + ' kg';
    $('f-wtv').textContent  = num(c.weight, 1) + ' kg';
    var rows = BANDS[c.sex].map(function(b){
      var hit = (b[0] === c.band);
      return '
' + b[0] + ''
        + '  ' + b[1] + '-' + (b[2] === 100 ? '+' : b[2]) + '%'
        + (hit ? '  ← you are here' : '') + '
';
    }).join('');
    $('f-bands').innerHTML = '
'
      + (c.sex === 'm' ? 'Male' : 'Female') + ' bands
' + rows;
    $('f-out').textContent = text(c);
  }

  function read(){
    return {
      sex: $('f-sex').value, weight: raw($('f-wt').value), height: raw($('f-ht').value),
      neck: raw($('f-neck').value), waist: raw($('f-waist').value), hip: raw($('f-hip').value)
    };
  }
  function go(){
    var c = compute(read());
    render(c);
    window.__T['body-fat-estimator'].last = c;
    return c;
  }

  ['f-sex','f-wt','f-ht','f-neck','f-waist','f-hip'].forEach(function(id){
    $(id).addEventListener('input', go);
    $(id).addEventListener('change', go);
  });
  $('f-go').addEventListener('click', go);
  $('f-cp').addEventListener('click', function(){
    navigator.clipboard.writeText(text(compute(read())));
  });
