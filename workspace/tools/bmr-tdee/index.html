  function num(x, d){ if(x === null || x === undefined || !isFinite(x)) return '-';
    return String(parseFloat(x.toFixed(d === undefined ? 2 : d))); }
  function raw(s){ var x = parseFloat(String(s).replace(/[^0-9.\-eE+]/g,'')); return isFinite(x) ? x : NaN; }
  function esc(s){ return String(s).replace(/&/g,'&').replace(/ protein  =  ' + num(c.protein_cal,0) + ' kcal
' +
      '
' + num(c.fat_g,1) + ' g fat       =  ' + num(c.fat_cal,0) + ' kcal
' +
      '
' + num(c.carb_g,1) + ' g carbs     =  ' + num(c.carb_cal,0) + ' kcal
' +
      '
Total = '
        + num(c.protein_cal + c.fat_cal + c.carb_cal, 0) + ' kcal
';
    $('b-out').textContent = text(c);
  }

  function text(c){
    return [
      'BMR AND TDEE',
      '-----------',
      'Sex            ' + (c.sex === 'm' ? 'Male' : 'Female'),
      'Weight         ' + num(c.kg,1) + ' kg',
      'Height         ' + num(c.cm,1) + ' cm',
      'Age            ' + num(c.age,0) + ' years',
      'Body fat       ' + num(c.bf,1) + ' %',
      'Lean mass      ' + num(c.lbm,1) + ' kg',
      '',
      'Activity       ' + c.activity + ' (x' + num(c.mult,3) + ')',
      'BMR (Mifflin)  ' + num(c.bmr_mifflin,0) + ' kcal/day',
      'BMR (Katch)    ' + num(c.bmr_katch,0) + ' kcal/day',
      'TDEE           ' + num(c.tdee,0) + ' kcal/day',
      '',
      'MACROS AT TDEE',
      'Protein        ' + num(c.protein_g,1) + ' g  = ' + num(c.protein_cal,0) + ' kcal',
      'Fat            ' + num(c.fat_g,1) + ' g  = ' + num(c.fat_cal,0) + ' kcal',
      'Carbohydrate   ' + num(c.carb_g,1) + ' g  = ' + num(c.carb_cal,0) + ' kcal'
    ].join('\n');
  }

  function read(){
    return {
      sex: $('b-sex').value, kg: raw($('b-kg').value), cm: raw($('b-cm').value),
      age: raw($('b-age').value), bf: raw($('b-bf').value), mult: parseFloat($('b-act').value)
    };
  }

  function go(){
    var c = compute(read());
    render(c);
    window.__T['bmr-tdee'].last = c;
    return c;
  }

  ['b-sex','b-kg','b-cm','b-age','b-bf','b-act'].forEach(function(id){
    $(id).addEventListener('input', go);
    $(id).addEventListener('change', go);
  });
  $('b-go').addEventListener('click', go);
  $('b-cp').addEventListener('click', function(){
    navigator.clipboard.writeText(text(compute(read())));
  });

  window.__T['bmr-tdee'] = { slug:'bmr-tdee', run: function(i){
    if(i.kg !== undefined) $('b-kg').value = i.kg;
    if(i.cm !== undefined) $('b-cm').value = i.cm;
    if(i.age !== undefined) $('b-age').value = i.age;
    if(i.bf !== undefined) $('b-bf').value = i.bf;
    if(i.mult !== undefined) $('b-act').value = String(i.mult);
    if(i.sex !== undefined) $('b-sex').value = i.sex;
    return JSON.stringify(go());
  }};
  go();
})();
