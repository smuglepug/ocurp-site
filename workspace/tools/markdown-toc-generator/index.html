Markdown TOC Generator

text — CLIENT-SIDE

Build a table of contents from Markdown headings with GitHub-accurate anchor slugs, including the -1 suffix on duplicate headings. ANCHORSDUPLICATE HANDLINGMARKDOWN + HTMLCLIENT-SIDE

Anchors use GitHub's rule: lowercase, spaces to hyphens, drop everything that is not a letter, digit or hyphen. Duplicate headings get the -1, -2 suffix GitHub appends. 


      

    

  



  

  


(function(){
  var $ = function(id){ return document.getElementById(id); };
  function slugifyHeading(text){
    return String(text).trim().toLowerCase()
      .replace(/[^\w\s-]/g, '')
      .replace(/\s+/g, '-');
  }
  function run(input){
    var lines = String(input == null ? '' : input).split(/\r?\n/);
    var heads = [], seen = {}, inFence = false;
    lines.forEach(function(line){
      if(/^\s*(```|~~~)/.test(line)){ inFence = !inFence; return; }
      if(inFence) return;
      var m = /^(#{1,6})\s+(.*)$/.exec(line);
      if(!m) return;
      var level = m[1].length;
      var text = m[2].replace(/\s+#+\s*$/,'').trim();
      var base = slugifyHeading(text);
      var slug = base, n = 0;
      while(Object.prototype.hasOwnProperty.call(seen, slug)){ n++; slug = base + '-' + n; }
      seen[slug] = true;
      heads.push({ level:level, text:text, slug:slug, indent:Math.min(level - 1, 5) });
    });
    var md = heads.map(function(h){
      return '  '.repeat(h.indent) + '- [' + h.text + '](#' + h.slug + ')';
    }).join('\n');
    return {
      count:heads.length,
      minLevel:heads.length ? Math.min.apply(null, heads.map(function(h){ return h.level; })) : 0,
      maxLevel:heads.length ? Math.max.apply(null, heads.map(function(h){ return h.level; })) : 0,
      headings:heads,
      slugs:heads.map(function(h){ return h.slug; }),
      duplicateSlugs:heads.length - Object.keys(seen).length,
      markdown:md,
      html:'
\n' + heads.map(function(h){
        return '  ' + h.text + '';
      }).join('\n') + '\n
'
    };
  }
  function render(){
    var r = run($('tc-in').value);
    $('tc-stats').innerHTML =
      '
Headings found' + r.count + '
' +
      '
Depth rangeH' + r.minLevel + ' – H' + r.maxLevel + '
' +
      '
Collisions suffixed' + r.duplicateSlugs + '
';
    $('tc-out').textContent = r.markdown;
    $('tc-html').textContent = r.html;
  }
  $('tc-go').addEventListener('click', render);
  $('tc-in').addEventListener('input', render);
  $('tc-cp').addEventListener('click', function(){ navigator.clipboard.writeText($('tc-out').textContent); });
  $('tc-cp2').addEventListener('click', function(){ navigator.clipboard.writeText($('tc-html').textContent); });
  render();
  window.__T = window.__T || {};
  window.__T['markdown-toc-generator'] = { slug:'markdown-toc-generator', run:run };
})();
